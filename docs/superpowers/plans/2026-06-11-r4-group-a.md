DO $$
DECLARE t text;
BEGIN
  FOREACH t IN ARRAY ARRAY['expense_categories','expenses','patient_relatives','stock_items','stock_movements','service_materials']
  LOOP
    EXECUTE format('ALTER TABLE %I ENABLE ROW LEVEL SECURITY', t);
    EXECUTE format('ALTER TABLE %I FORCE ROW LEVEL SECURITY', t);
    EXECUTE format($f$CREATE POLICY %I ON %I USING (tenant_id = NULLIF(current_setting('app.current_tenant_id', true), '')::uuid) WITH CHECK (tenant_id = NULLIF(current_setting('app.current_tenant_id', true), '')::uuid)$f$, t||'_tenant_isolation', t);
  END LOOP;
END $$;

async function deductForServices(c: Context, serviceIds: string[], sourceVisitNoteId: string) {
  if (serviceIds.length === 0) return;
  const db = getDb(c);
  const tenantId = c.get('activeTenantId');
  const boms = await db.select().from(serviceMaterials).where(inArray(serviceMaterials.serviceId, serviceIds));
  for (const bom of boms) {
    await db.insert(stockMovements).values({
      tenantId, itemId: bom.itemId, type: 'out', quantity: bom.qtyPerService,
      reason: 'visit_auto', sourceVisitNoteId, createdBy: c.get('user')?.id ?? null,
    });
    await db.update(stockItems)
      .set({ quantity: sql`${stockItems.quantity} - ${bom.qtyPerService}`, updatedAt: new Date() })
      .where(eq(stockItems.id, bom.itemId));
  }
}
