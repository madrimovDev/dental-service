```typescript
import type { PermissionDef, UserAssignment } from '@dental/api-client';
import type { PermissionGroup, UsersFilter } from './settings.types';

/**
 * Permission'larni kod PREFIKSI bo'yicha guruhlaydi (`code.split('.')[0]`).
 * Guruhlar resurs nomi bo'yicha alfabit; har guruh ichida code bo'yicha alfabit.
 * Sof funksiya — barqaror tartib (flaky emas).
 */
export function groupPermissionsByResource(perms: PermissionDef[]): PermissionGroup[] {
  const byResource = new Map<string, PermissionDef[]>();
  for (const p of perms) {
    const resource = p.code.split('.')[0] ?? p.code;
    const list = byResource.get(resource);
    if (list) list.push(p);
    else byResource.set(resource, [p]);
  }
  return [...byResource.entries()]
    .sort(([a], [b]) => a.localeCompare(b))
    .map(([resource, permissions]) => ({
      resource,
      permissions: [...permissions].sort((a, b) => a.code.localeCompare(b.code)),
    }));
}

/** Assignments'ni qidiruv (ism/telefon) + branch + rol bo'yicha filtrlaydi. Sof funksiya. */
export function filterAssignments(
  assignments: UserAssignment[],
  filter: UsersFilter,
): UserAssignment[] {
  const q = filter.search.trim().toLowerCase();
  return assignments.filter((a) => {
    if (q) {
      const name = (a.userFullName ?? '').toLowerCase();
      const phone = a.userPhone.toLowerCase();
      if (!name.includes(q) && !phone.includes(q)) return false;
    }
    if (filter.branchId && a.branchId !== filter.branchId) return false;
    if (filter.roleId && a.roleId !== filter.roleId) return false;
    return true;
  });
}
```

```typescript
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { api } from '@/lib/api.ts';
import { settingsKeys } from './settings.keys';

// --- Roles ---
export function useRoles() {
  return useQuery({ queryKey: settingsKeys.roles(), queryFn: api.roles.list });
}

export function useRole(id: string | null, enabled: boolean) {
  return useQuery({
    queryKey: settingsKeys.role(id ?? ''),
    queryFn: () => api.roles.get(id ?? ''),
    enabled: enabled && !!id,
  });
}

export function usePermissions() {
  return useQuery({ queryKey: settingsKeys.permissions(), queryFn: api.permissions.list });
}

export function useUpsertRole() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (input: {
      id: string | null;
      name: string;
      description?: string;
      permissions: string[];
    }) => {
      if (input.id) {
        await api.roles.update(input.id, { name: input.name, description: input.description });
        await api.roles.setPermissions(input.id, input.permissions);
        return;
      }
      await api.roles.create({
        name: input.name,
        description: input.description,
        permissions: input.permissions,
      });
    },
    onSuccess: () => qc.invalidateQueries({ queryKey: settingsKeys.roles() }),
  });
}

export function useDeleteRole() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (id: string) => api.roles.remove(id),
    onSuccess: () => qc.invalidateQueries({ queryKey: settingsKeys.roles() }),
  });
}

// --- Users / assignments ---
export function useUsers() {
  return useQuery({ queryKey: settingsKeys.users(), queryFn: api.users.listAssignments });

  const qc = useQueryClient();

    mutationFn: (body: { phone: string; fullName: string; roleId: string; branchId?: string }) =>
```

```tsx
import type { PermissionDef } from '@dental/api-client';
import { Card, Checkbox, Stack, Text } from '@mantine/core';
import { useTranslation } from 'react-i18next';
import { groupPermissionsByResource } from '../settings.mappers';

export function PermissionBuilder({
  permissions,
  selected,
  onChange,
}: {
  permissions: PermissionDef[];
  selected: Set<string>;
  onChange: (next: Set<string>) => void;
}) {
  const { t, i18n } = useTranslation();
  const groups = groupPermissionsByResource(permissions);

  const toggleOne = (code: string, checked: boolean) => {
    const next = new Set(selected);
    if (checked) next.add(code);
    else next.delete(code);
    onChange(next);
  };

  const toggleGroup = (codes: string[], checked: boolean) => {
    const next = new Set(selected);
    for (const code of codes) {
      if (checked) next.add(code);
      else next.delete(code);
    }
    onChange(next);
  };

  return (
    <Stack gap="sm">
      <Text fw={600} size="sm">
        {t('settings.perm_group_title')}
      </Text>
      {groups.map((group) => {
        const codes = group.permissions.map((p) => p.code);
        const allChecked = codes.every((c) => selected.has(c));
        const someChecked = codes.some((c) => selected.has(c));
        const label = t(`settings.perm_group.${group.resource}`, group.resource);
        return (
          <Card key={group.resource} withBorder padding="sm" radius="md">
            <Checkbox
              label={<Text fw={600} size="sm">{label}</Text>}
              checked={allChecked}
              indeterminate={!allChecked && someChecked}
              onChange={(e) => toggleGroup(codes, e.currentTarget.checked)}
              mb="xs"
            />
            <Stack gap="xs" pl="md">
              {group.permissions.map((p) => {
                const desc = i18n.language === 'ru' ? p.descriptionRu : p.descriptionUz;
                return (
                  <Checkbox
                    key={p.code}
                    label={
                      <Stack gap={0}>
                        <Text size="sm" ff="monospace">
                          {p.code}
                        </Text>
                        {desc && (
                          <Text size="xs" c="dimmed">
                            {desc}
                          </Text>
                        )}
                      </Stack>
                    }
                    checked={selected.has(p.code)}
                    onChange={(e) => toggleOne(p.code, e.currentTarget.checked)}
                  />
                );
              })}
            </Stack>
          </Card>
        );
      })}
    </Stack>
  );
}
```

```tsx
        )}
      </Stack>

      {/* Invite modal */}
      <Modal opened={opened} onClose={closeAll} title={t('users.invite')}>
        {tempPassword ? (
          <Stack gap="md">
            <Alert color="teal" variant="light">
              <Text fw={600}>{t('users.invited_successfully')}</Text>
              <Text size="sm">{t('users.temp_password_notice')}</Text>
              <Text mt="sm" fw={700} ff="monospace" size="lg">
                {tempPassword}
              </Text>
            </Alert>
            <Button onClick={closeAll} fullWidth>
              {t('common.cancel')}
            </Button>
          </Stack>
        ) : (
          <form onSubmit={form.onSubmit(submitInvite)}>
            <Stack gap="md">
              <Alert color="blue" variant="light">
                <Text size="sm">{t('users.invite_hint')}</Text>
              </Alert>
              <TextInput label={t('users.full_name')} required {...form.getInputProps('fullName')} />
              <PhoneInput label={t('users.phone')} required {...form.getInputProps('phone')} />
              <Select
                label={t('users.role')}
                data={roles.data?.map((r) => ({ value: r.id, label: r.name })) ?? []}
                required
                {...form.getInputProps('roleId')}
              />
              <Select
                label={t('users.branch')}
                description={t('users.branch_hint')}
                data={[
                  { value: '', label: t('users.all_branches') },
                  ...(branches.data?.map((b) => ({ value: b.id, label: b.name })) ?? []),
                ]}
                {...form.getInputProps('branchId')}
              />
              <Group justify="flex-end">
                <Button variant="default" onClick={closeAll}>
                  {t('common.cancel')}
                </Button>
                <Button type="submit" loading={invite.isPending}>
                  {t('users.send_invite')}
                </Button>
              </Group>
            </Stack>
          </form>
        )}
      </Modal>

      {/* Reset password result modal */}
      <Modal
        opened={!!resetResult}
        onClose={() => setResetResult(null)}
        title={t('users.reset_password')}
      >
        {resetResult && (
          <Stack gap="md">
            <Alert color="teal" variant="light">
              <Text fw={600}>{t('users.reset_password_success')}</Text>
              <Text size="sm" mt="xs">
                {resetResult.phone}
              </Text>
              <Text size="sm">{t('users.temp_password_notice')}</Text>
              <Text mt="sm" fw={700} ff="monospace" size="lg">
                {resetResult.tempPassword}
              </Text>
            </Alert>
            <Button onClick={() => setResetResult(null)} fullWidth>
              {t('common.cancel')}
            </Button>
          </Stack>
        )}
      </Modal>

      {/* Remove assignment confirm */}
      <ConfirmModal
        opened={!!toRemove}
        onClose={() => setToRemove(null)}
        onConfirm={() => {
          if (toRemove)
            removeAssignment.mutate(toRemove, { onSuccess: () => setToRemove(null) });
        }}
        message={t('users.confirm_remove')}
        loading={removeAssignment.isPending}
      />

      {/* Reset password confirm */}
      <ConfirmModal
        opened={!!toReset}
        onClose={() => setToReset(null)}
        onConfirm={() => {
          if (!toReset) return;
          resetPassword.mutate(toReset, {
            onSuccess: (data) => {
              setToReset(null);
              setResetResult({ phone: data.phone, tempPassword: data.tempPassword });
            },
            onError: (err) => {
              setToReset(null);
              notifications.show({
                title: t('common.error'),
                message: err instanceof ApiError ? err.message : 'Reset failed',
                color: 'red',
              });
            },
          });
        }}
        message={t('users.reset_password_confirm')}
        confirmColor="blue"
        loading={resetPassword.isPending}
      />
    </SettingsShell>
  );
}
```

```tsx
export function BranchesSettingsScreen() {

  const upsert = useUpsertBranch();

  const [opened, { open, close }] = useDisclosure(false);

          setEditing(null);
```

```tsx
import { ApiError } from '@dental/api-client';
import {
  Alert,
  Badge,
  Button,
  Card,
  Group,
  Loader,
  Modal,
  PasswordInput,
  Select,
  Stack,
  Text,
  TextInput,
} from '@mantine/core';
import { useForm } from '@mantine/form';
import { useDisclosure } from '@mantine/hooks';
import { notifications } from '@mantine/notifications';
import { zodResolver } from 'mantine-form-zod-resolver';
import { useEffect } from 'react';
import { useTranslation } from 'react-i18next';
import { useAuthStore } from '@/stores/auth-store.ts';
import {
  useTenant,
  useTransferOwnership,
  useUpdateTenant,
  useUsers,
} from '../settings.api';
import {
  type TenantFormValues,
  type TransferFormValues,
  tenantSchema,
  transferSchema,
} from '../settings.schema';
import { ConfirmModal } from './ConfirmModal';
import { SettingsShell } from './SettingsShell';

export function TenantSettingsScreen() {
  const { t } = useTranslation();
  const { hasPermission } = useAuthStore();
  const tenant = useTenant();
  const update = useUpdateTenant();
  const transfer = useTransferOwnership();
  const users = useUsers();
  const [transferOpened, { open: openTransfer, close: closeTransfer }] = useDisclosure(false);
  const [confirmOpened, { open: openConfirm, close: closeConfirm }] = useDisclosure(false);

  const form = useForm<TenantFormValues>({
    initialValues: { name: '', defaultLanguage: 'uz' },
    validate: zodResolver(tenantSchema),
  });

  const transferForm = useForm<TransferFormValues>({
    initialValues: { newOwnerUserId: '', currentPassword: '' },
    validate: zodResolver(transferSchema),
  });

  // biome-ignore lint/correctness/useExhaustiveDependencies: form.setValues stable
  useEffect(() => {
    if (tenant.data) {
      form.setValues({
        name: tenant.data.name,
        defaultLanguage: (tenant.data.defaultLanguage as 'uz' | 'ru') ?? 'uz',
      });
    }
  }, [tenant.data]);

  if (tenant.isLoading) {
    return (
      <SettingsShell active="tenant">
        <Loader />
      </SettingsShell>
    );
  }

  const data = tenant.data;

  // Egasiga o'tkazish uchun nomzodlar (unique userId, joriy egadan tashqari).
  const ownerCandidates = [
    ...new Map(
      (users.data ?? [])
        .filter((a) => a.userId !== data?.ownerUserId)
        .map((a) => [a.userId, a]),
    ).values(),
  ].map((a) => ({ value: a.userId, label: `${a.userFullName ?? a.userPhone} (${a.userPhone})` }));

  const doTransfer = (v: TransferFormValues) => {
    transfer.mutate(v, {
      onSuccess: () => {
        closeConfirm();
        closeTransfer();
        transferForm.reset();
      },
      onError: (err) => {
        closeConfirm();
        notifications.show({
          title: t('common.error'),
          message: err instanceof ApiError ? err.message : 'Transfer failed',
          color: 'red',
        });
      },
    });
  };

  return (
    <SettingsShell active="tenant">
      <Stack gap="lg">
        {!hasPermission('tenant.edit') && (
          <Alert color="yellow" variant="light">
            {t('tenant_settings.readonly_notice')}
          </Alert>
        )}

        <Card withBorder padding="lg" radius="md">
          <Stack gap="md">
            <Group justify="space-between">
              <Stack gap={2}>
                <Text size="sm" c="dimmed">
                  {t('tenant_settings.slug')}
                </Text>
                <Text fw={600}>{data?.slug ?? '—'}</Text>
              </Stack>
              <Badge color={data?.subscriptionStatus === 'active' ? 'teal' : 'yellow'}>
                {data?.subscriptionStatus}
              </Badge>
            </Group>

            <form onSubmit={form.onSubmit((values) => update.mutate(values))}>
              <Stack gap="md">
                <TextInput
                  label={t('tenant_settings.name')}
                  required
                  disabled={!hasPermission('tenant.edit')}
                  {...form.getInputProps('name')}
                />
                <Select
                  label={t('tenant_settings.default_language')}
                  data={[
                    { value: 'uz', label: "O'zbek" },
                    { value: 'ru', label: 'Русский' },
                  ]}
                  disabled={!hasPermission('tenant.edit')}
                  {...form.getInputProps('defaultLanguage')}
                />
                {hasPermission('tenant.edit') && (
                  <Group justify="flex-end">
                    <Button type="submit" loading={update.isPending}>
                      {t('common.save')}
                    </Button>
                  </Group>
                )}
              </Stack>
            </form>
          </Stack>
        </Card>

        {hasPermission('tenant.edit') && (
          <Card withBorder padding="lg" radius="md">
            <Stack gap="md">
              <Text fw={600}>{t('settings.transfer_ownership')}</Text>
              <Text size="sm" c="dimmed">
                {t('settings.transfer_hint')}
              </Text>
              <Group>
                <Button color="red" variant="light" onClick={openTransfer}>
                  {t('settings.transfer_ownership')}
                </Button>
              </Group>
            </Stack>
          </Card>
        )}
      </Stack>

      {/* Transfer form modal */}
      <Modal
        opened={transferOpened}
        onClose={closeTransfer}
        title={t('settings.transfer_ownership')}
      >
        <Stack gap="md">
          <Select
            label={t('settings.transfer_new_owner')}
            data={ownerCandidates}
            searchable
            required
            {...transferForm.getInputProps('newOwnerUserId')}
          />
          <PasswordInput
            label={t('settings.transfer_current_password')}
            required
            {...transferForm.getInputProps('currentPassword')}
          />
          <Group justify="flex-end">
            <Button variant="default" onClick={closeTransfer}>
              {t('common.cancel')}
            </Button>
            <Button
              color="red"
              onClick={() => {
                if (!transferForm.validate().hasErrors) openConfirm();
              }}
            >
              {t('settings.transfer_ownership')}
            </Button>
          </Group>
        </Stack>
      </Modal>

      {/* Transfer confirm */}
      <ConfirmModal
        opened={confirmOpened}
        onClose={closeConfirm}
        onConfirm={() => doTransfer(transferForm.values)}
        message={t('settings.transfer_confirm')}
        loading={transfer.isPending}
      />
    </SettingsShell>
  );
}
```
