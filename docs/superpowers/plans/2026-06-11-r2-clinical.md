Loading holati yoki anchor kerak bo'lganda **Mantine `Button`** (themed teal) ishlatiladi. Static action button'larda R0 `Button` mumkin, lekin izchillik uchun bu re-skinda **Mantine `Button` (themed)** ishlatamiz — etalon queue ham shunday qiladi. 3. **Ikonkalar = `@tabler/icons-react` (lucide EMAS).** ⚠️ **Spec matni va task quality-rule "lucide" deydi, lekin:** (a) lucide-react@0.469 da **`Tooth`/dental ikonka YO'Q** (`grep` bilan tekshirildi — `index_Tooth` eksport yo'q); tabler'da `IconDental` bor.   const map: ToothStateMap = new Map();

```typescript
    if (s.surface === 'whole') map.set(s.toothNumber, s.state);
  }
  return map;
}
```

```tsx
import { IconDental } from '@tabler/icons-react';
import { useEffect, useMemo, useState } from 'react';
import { useTranslation } from 'react-i18next';
import { useAuthStore } from '@/stores/auth-store.ts';
import { useCreateToothChart, useLatestToothChart } from '../../clinical.api';

export function OdontogramTab({ patientId }: Props) {
  const { t } = useTranslation();
  const { hasPermission } = useAuthStore();
  const canEdit = hasPermission('medical_record.edit');

  const chartQuery = useLatestToothChart(patientId);
  const createMut = useCreateToothChart(patientId);

  const [isPrimary, setIsPrimary] = useState(false);
  const [selectedTooth, setSelectedTooth] = useState<number | null>(null);
  const [draftStates, setDraftStates] = useState<Map<number, ToothState>>(new Map());
  const [draftNotes, setDraftNotes] = useState('');
  const [dirty, setDirty] = useState(false);

  useEffect(() => {
    if (!chartQuery.data) return;
    setDraftStates(toToothStateMap(chartQuery.data));
    setIsPrimary(chartQuery.data.isPrimaryDentition);
    setDirty(false);
  }, [chartQuery.data]);

  const statesArray: ToothStateEntry[] = useMemo(
    () =>
      Array.from(draftStates.entries()).map(([toothNumber, state]) => ({
        id: `draft-${toothNumber}`,
        toothChartId: 'draft',
        toothNumber,
        surface: 'whole' as const,
        state,
        diagnosisId: null,
        notes: null,
        createdAt: new Date().toISOString(),
      })),
    [draftStates],
  );

  const setToothState = (state: ToothState) => {
    if (selectedTooth === null) return;
    const next = new Map(draftStates);
    next.set(selectedTooth, state);
    setDraftStates(next);
    setDirty(true);
  };

  const save = () => {
    const states: CreateToothStateInput[] = draftStatesToCreateInput(draftStates);
    createMut.mutate(
      { patientId, isPrimaryDentition: isPrimary, notes: draftNotes || undefined, states },
      {
        onSuccess: () => {
          setDraftNotes('');
          setDirty(false);
        },
      },
    );
  };

  return (
    <Stack gap="md">
      <Group justify="space-between">
        <Title order={4}>{t('tooth_chart.title')}</Title>
        <Switch
          label={t('tooth_chart.primary_dentition')}
          checked={isPrimary}
          onChange={(e) => {
            setIsPrimary(e.currentTarget.checked);
            setDraftStates(new Map());
            setDirty(true);
          }}
          disabled={!canEdit}
        />
      </Group>

      {!chartQuery.data && !chartQuery.isLoading && (
        <Alert color="gray">{t('tooth_chart.no_chart')}</Alert>
      )}

      <Paper withBorder p="md">
        <Odontogram
          states={statesArray}
          isPrimaryDentition={isPrimary}
          selectedTooth={selectedTooth}
          onSelectTooth={setSelectedTooth}
        />
      </Paper>

      {canEdit && selectedTooth !== null && (
        <Paper withBorder p="md">
          <Stack gap="sm">
            <Group gap={6}>
              <IconDental size={18} />
              <Text fw={600}>
                {t('tooth_chart.selected_tooth')}: <strong>{selectedTooth}</strong>
              </Text>
            </Group>
            <Select
              label={t('tooth_chart.set_state')}
              value={draftStates.get(selectedTooth) ?? 'healthy'}
              onChange={(v) => v && setToothState(v as ToothState)}
              data={TOOTH_STATE_OPTIONS.map((s) => ({
                value: s,
                label: t(`tooth_chart.states.${s}`),
              }))}
            />
          </Stack>
        </Paper>
      )}

      {canEdit && (
        <Paper withBorder p="md">
          <Stack gap="sm">
            <Textarea
              label={t('treatment_plan.notes')}
              placeholder={t('tooth_chart.notes_placeholder')}
              value={draftNotes}
              onChange={(e) => setDraftNotes(e.currentTarget.value)}
              rows={2}
            />
            <Group justify="flex-end">
              <Button onClick={save} loading={createMut.isPending} disabled={!dirty && draftStates.size === 0}>
                {t('tooth_chart.save_snapshot')}
              </Button>
            </Group>
          </Stack>
        </Paper>
      )}
    </Stack>
  );
}
```

```tsx
export function PrescriptionsTab({ patientId }: Props) {
```

```tsx
import type { PatientImageType } from '@dental/api-client';
import { Button, FileInput, Group, Modal, Select, Stack, TextInput, Textarea } from '@mantine/core';
import { useForm } from '@mantine/form';
import { zodResolver } from 'mantine-form-zod-resolver';
import { useState } from 'react';
import { useTranslation } from 'react-i18next';
import { useUploadPatientImage } from '../../clinical.api';
import { type UploadImageFormValues, uploadImageFormSchema } from '../../clinical.schema';

const TYPE_OPTIONS: PatientImageType[] = [
  'xray_panoramic', 'xray_periapical', 'xray_bitewing', 'xray_cbct',
  'intraoral_photo', 'face_photo', 'before_after', 'other',
];

export function ImageUploadModal({ opened, onClose, patientId }: { opened: boolean; onClose: () => void; patientId: string }) {
  const { t } = useTranslation();
  const [file, setFile] = useState<File | null>(null);
  const uploadMut = useUploadPatientImage(patientId);
  const form = useForm<UploadImageFormValues>({
    initialValues: { type: 'intraoral_photo', title: '', description: '', toothNumbersText: '', takenDate: '' },
    validate: zodResolver(uploadImageFormSchema),
  });

  const handleClose = () => { form.reset(); setFile(null); onClose(); };

  const submit = form.onSubmit((values) => {
    if (!file) return;
    const fd = new FormData();
    fd.append('file', file);
    fd.append('patientId', patientId);
    fd.append('type', values.type);
    if (values.title) fd.append('title', values.title);
    if (values.description) fd.append('description', values.description);
    if (values.toothNumbersText) fd.append('toothNumbers', values.toothNumbersText);
    if (values.takenDate) fd.append('takenDate', new Date(values.takenDate).toISOString());
    uploadMut.mutate(fd, { onSuccess: handleClose });
  });

  return (
    <Modal opened={opened} onClose={handleClose} title={t('image.upload')} size="lg">
      <form onSubmit={submit}>
        <Stack gap="md">
```

```tsx
import { Button, Group, Modal, Paper, SimpleGrid, Stack, Text, Title } from '@mantine/core';
import { useDisclosure } from '@mantine/hooks';
import { IconPhotoPlus } from '@tabler/icons-react';
import { useMemo, useState } from 'react';
import { useTranslation } from 'react-i18next';
import { useAuthStore } from '@/stores/auth-store.ts';
import { useDeletePatientImage, usePatientImages } from '../../clinical.api';
import { toPatientImageVM } from '../../clinical.mappers';
import type { PatientImageVM } from '../../clinical.types';
import { ImageCard } from './ImageCard';
import { ImagePreviewModal } from './ImagePreviewModal';
import { ImageUploadModal } from './ImageUploadModal';

type Props = { patientId: string };

export function PatientImagesTab({ patientId }: Props) {
  const { t } = useTranslation();
  const { hasPermission } = useAuthStore();
  const [uploadOpened, { open: openUpload, close: closeUpload }] = useDisclosure(false);
  const [previewId, setPreviewId] = useState<string | null>(null);
  const [deleteId, setDeleteId] = useState<string | null>(null);

  const canUpload = hasPermission('patient_image.upload');
  const canView = hasPermission('patient_image.view');

  const listQuery = usePatientImages(patientId);
  const deleteMut = useDeletePatientImage(patientId);

  const vms: PatientImageVM[] = useMemo(
    () => (listQuery.data ?? []).map(toPatientImageVM),
    [listQuery.data],
  );

  if (!canView) return <Text c="dimmed">{t('patients.no_medical_permission')}</Text>;

  const previewImage = vms.find((v) => v.id === previewId) ?? null;

  return (
    <Stack gap="md">
      <Group justify="space-between">
        <Title order={4}>{t('image.title')}</Title>
        {canUpload && (
          <Button leftSection={<IconPhotoPlus size={16} />} onClick={openUpload}>
            {t('image.upload')}
          </Button>
        )}
      </Group>

      {listQuery.isLoading && <Text>{t('common.loading')}</Text>}
      {!listQuery.isLoading && vms.length === 0 && (
        <Paper withBorder p="md"><Text c="dimmed">{t('image.empty')}</Text></Paper>
      )}

      <SimpleGrid cols={{ base: 1, sm: 2, md: 3, lg: 4 }} spacing="md">
        {vms.map((vm) => (
          <ImageCard
            key={vm.id}
            vm={vm}
            canUpload={canUpload}
            onPreview={() => setPreviewId(vm.id)}
            onRequestDelete={() => setDeleteId(vm.id)}
          />
        ))}
      </SimpleGrid>

      <ImagePreviewModal image={previewImage} onClose={() => setPreviewId(null)} />
      <ImageUploadModal opened={uploadOpened} onClose={closeUpload} patientId={patientId} />

      {/* Delete confirm — Mantine modal (native confirm() o'rniga) */}
      <Modal opened={!!deleteId} onClose={() => setDeleteId(null)} title={t('image.confirm_delete')} size="sm">
        <Group justify="flex-end" mt="md">
          <Button variant="default" onClick={() => setDeleteId(null)}>{t('common.cancel')}</Button>
          <Button color="red" loading={deleteMut.isPending}
            onClick={() => {
              if (deleteId) deleteMut.mutate(deleteId, { onSuccess: () => setDeleteId(null) });
            }}>
            {t('common.delete')}
          </Button>
        </Group>
      </Modal>
    </Stack>
  );
}
```
