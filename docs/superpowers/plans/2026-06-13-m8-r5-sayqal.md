```ts
export const VERTICALS = ['dental', 'general'] as const;
export type Vertical = (typeof VERTICALS)[number];

export type VerticalFeature = 'odontogram' | 'labOrders';

export const VERTICAL_FEATURES: Record<Vertical, Record<VerticalFeature, boolean>> = {
  dental: { odontogram: true, labOrders: true },
  general: { odontogram: false, labOrders: false },
};

/** Berilgan vertical mazkur feature'ni yoqadimi. Noma'lum vertical → false. */
export const hasFeature = (vertical: Vertical, feature: VerticalFeature): boolean =>
  VERTICAL_FEATURES[vertical]?.[feature] ?? false;
```

`packages/ui/src/primitives/StateView/StateView.tsx`:
```tsx
import { Skeleton } from '@mantine/core';
import type { ReactNode } from 'react';
import classes from './StateView.module.css';

export type StateViewStatus = 'loading' | 'error' | 'empty';
export type StateViewSkeleton = 'table' | 'cards' | 'detail';

export interface StateViewProps {
  status: StateViewStatus;
  /** loading uchun skeleton shakli */
  skeleton?: StateViewSkeleton;
  /** empty/error uchun */
  icon?: ReactNode;
  title?: string;
  description?: string;
  action?: ReactNode;
  /** error uchun */
  onRetry?: () => void;
  retryLabel?: string;
}

function SkeletonBlock({ shape }: { shape: StateViewSkeleton }) {
  if (shape === 'detail') {
    return (
      <div className={classes.skeletonWrap}>
        <Skeleton height={40} width="40%" radius="md" />
        <Skeleton height={120} radius="md" mt="md" />
        <Skeleton height={120} radius="md" mt="sm" />
      </div>
    );
  }
  if (shape === 'cards') {
    return (
      <div className={classes.cards}>
        {[0, 1, 2, 3].map((i) => (
          <Skeleton key={i} height={96} radius="md" />
        ))}
      </div>
    );
  }
  return (
    <div className={classes.skeletonWrap}>
      {[0, 1, 2, 3, 4].map((i) => (
        <Skeleton key={i} height={44} radius="sm" mt={i === 0 ? 0 : 8} />
      ))}
    </div>
  );
}

export function StateView(props: StateViewProps) {
  const { status, skeleton = 'table', icon, title, description, action, onRetry, retryLabel } = props;

  if (status === 'loading') {
    return (
      <div className={classes.root} data-state="loading" aria-busy="true">
        <SkeletonBlock shape={skeleton} />
      </div>
    );
  }

  return (
    <div className={classes.root} data-state={status}>
      <div className={classes.center}>
        {icon && <div className={classes.icon}>{icon}</div>}
        {title && <div className={classes.title}>{title}</div>}
        {description && <div className={classes.description}>{description}</div>}
        {status === 'error' && onRetry && (
          <button type="button" className={classes.retry} onClick={onRetry}>
            {retryLabel ?? 'Retry'}
          </button>
        )}
        {action && <div className={classes.action}>{action}</div>}
      </div>
    </div>
  );
}
```

```ts
import { hasFeature, type Vertical, type VerticalFeature } from '@dental/core';
import type { MiddlewareHandler } from 'hono';
import { tenantSettings } from '@dental/db-schema';
import { getDb } from '../db/scoped';
import { fail } from '../lib/response';

/**
 * Joriy tenant verticali mazkur feature'ni yoqmasa 403 qaytaradi.
 * UI yashirsa-da API ham yopiq bo'lishi uchun (defense-in-depth, CLI-153 saboq).
 */
export const requireFeature =
  (feature: VerticalFeature): MiddlewareHandler =>
  async (c, next) => {
    const db = getDb(c);
    const [settings] = await db.select().from(tenantSettings).limit(1);
    const vertical = (settings?.vertical ?? 'dental') as Vertical;
    if (!hasFeature(vertical, feature)) {
      return c.json(fail('feature.disabled', `Feature not available for vertical: ${feature}`), 403);
    }
    await next();
  };
```

> `fail` helper'ning aniq importini mavjud middleware (`require-permission.ts`) dan ko'chir. `tenantSettings` importini db-schema barrelidan tasdiqla.

```ts
import { hasFeature, type Vertical, type VerticalFeature } from '@dental/core';

export type NavItem = {
  key: string;
  labelKey: string;
  icon: LucideIcon;
  path: string;
  permission?: string | string[];
  /** Vertical feature gate — vertical yoqmasa item tushadi. */
  feature?: VerticalFeature;
  disabled?: true;
};

function isVisible(item: NavItem, has: (p: string) => boolean, vertical: Vertical): boolean {
  if (item.feature && !hasFeature(vertical, item.feature)) return false;
  if (item.disabled) return true;
  if (!item.permission) return true;
  const perms = Array.isArray(item.permission) ? item.permission : [item.permission];
  return perms.some((p) => has(p));
}

export function visibleGroups(has: (p: string) => boolean, vertical: Vertical = 'dental'): NavGroup[] {
  return NAV_GROUPS.map((g) => ({ ...g, items: g.items.filter((i) => isVisible(i, has, vertical)) })).filter(
    (g) => g.items.length > 0,
  );
}
```

Agar nav'da odontogram/lab item bo'lsa, ularga `feature: 'odontogram'` / `feature: 'labOrders'` qo'sh.

```tsx
import { Spotlight, type SpotlightActionData, type SpotlightActionGroupData } from '@mantine/spotlight';
import { useDebouncedValue } from '@mantine/hooks';
import { Search } from 'lucide-react';
import { useMemo, useState } from 'react';
import { useTranslation } from 'react-i18next';
import { useNavigate } from 'react-router-dom';

import { buildQuickActions } from './command-actions';
import { visibleGroups } from './nav-config';

export function CommandPalette() {
  const { t } = useTranslation();
  const navigate = useNavigate();
  const { hasPermission } = useAuthStore();
  const { data: settings } = useTenantSettings();
  const vertical = (settings?.vertical ?? 'dental') as Vertical;

  const [query, setQuery] = useState('');
  const [debounced] = useDebouncedValue(query, 250);
  const canSearchPatients = hasPermission('patient.view');
  const patientsQuery = usePatients({
    search: debounced,
    archived: false,
    page: 1,
    pageSize: 6,
  });
  const showPatients = canSearchPatients && debounced.trim().length >= 2;

  const groups = useMemo<(SpotlightActionGroupData | SpotlightActionData)[]>(() => {
    const navItems: SpotlightActionData[] = visibleGroups(hasPermission, vertical)
      .flatMap((g) => g.items)
      .filter((i) => !i.disabled)
      .map((i) => ({ id: `nav-${i.key}`, label: t(i.labelKey), onClick: () => navigate(i.path) }));

    const actions: SpotlightActionData[] = buildQuickActions({ t, navigate, has: hasPermission }).map(
      (a) => ({ id: a.id, label: a.label, onClick: a.onTrigger }),
    );

    const patientActions: SpotlightActionData[] = showPatients
      ? (patientsQuery.data?.items ?? []).map((p) => ({
          id: `patient-${p.id}`,
          label: p.fullName,
          description: p.phone,
          onClick: () => navigate(`/patients/${p.id}`),
        }))
      : [];

    const result: (SpotlightActionGroupData)[] = [
      { group: t('palette.group.navigation'), actions: navItems },
      { group: t('palette.group.actions'), actions },
    ];
    if (canSearchPatients) {
      result.push({ group: t('palette.group.patients'), actions: patientActions });
    }
    return result;
  }, [hasPermission, vertical, t, navigate, showPatients, patientsQuery.data, canSearchPatients]);

  return (
    <Spotlight
      actions={groups}
      query={query}
      onQueryChange={setQuery}
      shortcut={['mod + K']}
      nothingFound={t('palette.nothing_found')}
      searchProps={{ leftSection: <Search size={18} />, placeholder: t('shell.search_placeholder') }}
    />
  );
}
```

```tsx
import { Card } from '@dental/ui';

import { Check, ChevronRight, X } from 'lucide-react';
import { useTranslation } from 'react-i18next';
import { useNavigate } from 'react-router-dom';
import { useBranches, useTenantSettings, useUpdateTenantSettings, useUsers } from '@/features/settings/settings.api.ts';

import { useAuthStore } from '@/stores/auth-store.ts';
import { buildOnboardingSteps, isOnboardingComplete } from '../onboarding';

export function OnboardingChecklist() {
  const { t } = useTranslation();
  const navigate = useNavigate();
  const { hasPermission } = useAuthStore();
  const { data: settings } = useTenantSettings();
  const update = useUpdateTenantSettings();
  const branches = useBranches();
  const services = useServices();
  const users = useUsers();

  // Faqat sozlash huquqi borlar; dismiss qilingan bo'lsa yashir

    staff: users.data?.length ?? 0,
  });

  const doneCount = steps.filter((s) => s.done).length;
  const dismiss = () => update.mutate({ onboardingDismissedAt: new Date().toISOString() });

  return (

      <Group justify="space-between" mb="xs">

        <ActionIcon variant="subtle" aria-label={t('onboarding.dismiss')} onClick={dismiss}>
          <X size={16} />
        </ActionIcon>
      </Group>
      <Text size="sm" c="dimmed" mb="sm">{t('onboarding.subtitle')}</Text>
      <Progress value={(doneCount / steps.length) * 100} mb="md" />
      <Stack gap="xs">
        {steps.map((s) => (

            key={s.id}
            justify="space-between"
            onClick={() => !s.done && navigate(s.path)}
            style={{ cursor: s.done ? 'default' : 'pointer' }}

                <Text size="xs" c="dimmed">{s.description}</Text>
              </div>
            </Group>
          </Group>
        ))}
      </Stack>
    </Card>
  );
}
```

```tsx
import { StateView } from '@dental/ui';
import { AlertTriangle } from 'lucide-react';
import { Component, type ErrorInfo, type ReactNode } from 'react';
import { withTranslation, type WithTranslation } from 'react-i18next';

type Props = WithTranslation & { children: ReactNode };
type State = { hasError: boolean };

class ErrorBoundaryInner extends Component<Props, State> {
  state: State = { hasError: false };

  static getDerivedStateFromError(): State {
    return { hasError: true };
  }

  componentDidCatch(error: Error, info: ErrorInfo) {
    console.error('ErrorBoundary caught:', error, info);
  }

  render() {
    const { t, children } = this.props;
    if (this.state.hasError) {
      return (
        <StateView
          status="error"
          icon={<AlertTriangle size={32} />}
          title={t('errors.boundary_title')}
          description={t('errors.boundary_desc')}
          retryLabel={t('errors.reload')}
          onRetry={() => window.location.reload()}
        />
      );
    }
    return children;
  }
}

export const ErrorBoundary = withTranslation()(ErrorBoundaryInner);
```
