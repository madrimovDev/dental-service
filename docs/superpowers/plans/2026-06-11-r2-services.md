    const db = getDb(c);
    const [before] = await db
      .select()
      .from(services)
      .where(and(eq(services.id, id), isNull(services.deletedAt)));
    if (!before) return null;

    await db
      .update(services)
      .set({ deletedAt: new Date(), updatedAt: new Date() })

      resourceType: 'service',
      resourceId: id,
      before,
    });

    return before;

```typescript
import type { Service } from '@dental/api-client';
import type { CategoryGroup, ServiceVM } from './services.types';

export function toServiceVM(s: Service): ServiceVM {
  return {
    id: s.id,
    code: s.code,
    nameUz: s.nameUz,
    nameRu: s.nameRu,
    category: s.category,
    durationMinutes: s.durationMinutes,
    isActive: s.isActive,
  };
}

/** Nom (uz/ru) yoki kod bo'yicha case-insensitive qidiruv + kategoriya filter (sof). */
export function filterServices(
  list: Service[],
  search: string,
  category: string | null,
): Service[] {
  const q = search.trim().toLowerCase();
  return list.filter((s) => {
    if (category !== null && s.category !== category) return false;
    if (!q) return true;
    return (
      s.nameUz.toLowerCase().includes(q) ||
      (s.nameRu?.toLowerCase().includes(q) ?? false) ||
      s.code.toLowerCase().includes(q)
    );
  });
}

/**
 * Kategoriya bo'yicha guruhlash. Kategoriyalar alifbo tartibida,
 * kategoriyasiz (null) guruh oxirida. Guruh ichida kirish tartibi saqlanadi.
 */
export function groupByCategory(list: Service[]): CategoryGroup[] {
  const byCategory = new Map<string | null, ServiceVM[]>();
  for (const s of list) {
    const key = s.category;
    const bucket = byCategory.get(key);
    if (bucket) bucket.push(toServiceVM(s));
    else byCategory.set(key, [toServiceVM(s)]);
  }

  const named = [...byCategory.keys()].filter((k): k is string => k !== null).sort((a, b) =>
    a.localeCompare(b),
  );
  const ordered: (string | null)[] = byCategory.has(null) ? [...named, null] : named;

  return ordered.map((category) => ({
    category,
    services: byCategory.get(category) ?? [],
  }));
}
```

import { serviceKeys } from './services.keys';
import type { ServiceFormValues } from './services.types';

export function useServiceDetail(id: string | null, enabled = true) {
  return useQuery({
    queryKey: serviceKeys.detail(id ?? ''),
    queryFn: () => api.services.get(id ?? ''),
    enabled: !!id && enabled,
  });

export function useBranches() {
  return useQuery({ queryKey: serviceKeys.branches(), queryFn: api.branches.list });

/** Create yoki update (editing bo'lsa update). */
export function useUpsertService(editing: Service | null) {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (values: ServiceFormValues) => {
      if (editing) {
        return api.services.update(editing.id, {
          nameUz: values.nameUz,
          nameRu: values.nameRu || null,
          category: values.category || null,
          durationMinutes: values.durationMinutes,
          requiredRole: values.requiredRole || null,
          descriptionUz: values.descriptionUz || null,
          descriptionRu: values.descriptionRu || null,
          isActive: values.isActive,
        });
      }
      return api.services.create({
        code: values.code,
        nameUz: values.nameUz,
        nameRu: values.nameRu || undefined,
        category: values.category || undefined,
        durationMinutes: values.durationMinutes,
        requiredRole: values.requiredRole || undefined,
        descriptionUz: values.descriptionUz || undefined,
        descriptionRu: values.descriptionRu || undefined,
        isActive: values.isActive,
        initialBranchId: values.initialBranchId || undefined,
        initialPrice: values.initialPrice,
      });
    },
    onSuccess: () => qc.invalidateQueries({ queryKey: serviceKeys.all }),
  });
}

export function useSetPrice(serviceId: string | null) {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: ({ branchId, price }: { branchId: string; price: number }) =>
      api.services.setPrice(serviceId ?? '', { branchId, price }),
    onSuccess: () => {
      if (serviceId) void qc.invalidateQueries({ queryKey: serviceKeys.detail(serviceId) });
    },
  });
}

export function useDeleteService() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (id: string) => api.services.remove(id),
    onSuccess: () => qc.invalidateQueries({ queryKey: serviceKeys.all }),
  });
}

            {prices.map((p) => (

          label={t('services.branch')}

          onChange={setBranchId}
          clearable
          flex={1}
        />
        <NumberInput
          label={t('services.price')}
          min={0}
          step={1000}
          suffix=" so'm"
          value={price}

              setBranchId(null);

import type { Service } from '@dental/api-client';
import { Button, Group, Modal, Stack, Tabs } from '@mantine/core';
import { useForm } from '@mantine/form';
import { zodResolver } from 'mantine-form-zod-resolver';
import { useEffect } from 'react';
import { useTranslation } from 'react-i18next';
import { useBranches, useServiceDetail, useSetPrice, useUpsertService } from '../services.api';
import { serviceFormSchema } from '../services.schema';
import type { ServiceFormValues } from '../services.types';
import { ServiceInfoTab } from './ServiceInfoTab';
import { ServicePricesTab } from './ServicePricesTab';

const EMPTY: ServiceFormValues = {
  code: '',
  nameUz: '',
  nameRu: '',
  category: '',
  durationMinutes: 30,
  requiredRole: '',
  descriptionUz: '',
  descriptionRu: '',
  isActive: true,
  initialBranchId: undefined,
  initialPrice: undefined,
};

export function ServiceFormModal({
  opened,
  onClose,
  editing,
}: {
  opened: boolean;
  onClose: () => void;
  editing: Service | null;
}) {
  const { t } = useTranslation();
  const isEditing = !!editing;

  const branches = useBranches();
  const detail = useServiceDetail(editing?.id ?? null, opened);
  const upsert = useUpsertService(editing);
  const setPrice = useSetPrice(editing?.id ?? null);

  const form = useForm<ServiceFormValues>({
    initialValues: EMPTY,
    validate: zodResolver(serviceFormSchema),
  });

  // editing o'zgarganda formani to'ldirish / tozalash
  useEffect(() => {
    if (!opened) return;
    if (editing) {
      form.setValues({
        ...EMPTY,
        code: editing.code,
        nameUz: editing.nameUz,
        nameRu: editing.nameRu ?? '',
        category: editing.category ?? '',
        durationMinutes: editing.durationMinutes,
        requiredRole: editing.requiredRole ?? '',
        descriptionUz: editing.descriptionUz ?? '',
        descriptionRu: editing.descriptionRu ?? '',
        isActive: editing.isActive,
      });
    } else {
      form.setValues(EMPTY);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [opened, editing?.id]);

  const handleClose = () => {
    form.reset();
    onClose();
  };

  return (
    <Modal
      opened={opened}
      onClose={handleClose}
      title={isEditing ? `${t('services.edit')}: ${editing.code}` : t('services.create')}
      size="lg"
    >
      <Tabs defaultValue="info">
        <Tabs.List mb="md">
          <Tabs.Tab value="info">{t('services.tab_info')}</Tabs.Tab>
          <Tabs.Tab value="prices">{t('services.tab_prices')}</Tabs.Tab>
        </Tabs.List>

        <Tabs.Panel value="info">
          <form
            onSubmit={form.onSubmit((values) =>
              upsert.mutate(values, { onSuccess: handleClose }),
            )}
          >
            <Stack gap="md">
              <ServiceInfoTab form={form} isEditing={isEditing} />
              <Group justify="flex-end">
                <Button variant="default" onClick={handleClose}>
                  {t('common.cancel')}
                </Button>
                <Button type="submit" loading={upsert.isPending}>
                  {t('common.save')}
                </Button>
              </Group>
            </Stack>
          </form>
        </Tabs.Panel>

        <Tabs.Panel value="prices">
          <ServicePricesTab
            isEditing={isEditing}
            prices={detail.data?.prices ?? []}
            branches={branches.data ?? []}
            onSetPrice={(branchId, price) => setPrice.mutate({ branchId, price })}

import type { Service } from '@dental/api-client';
import { Button, Container, Group, Modal, Select, Stack, Text, TextInput } from '@mantine/core';
import { useDisclosure } from '@mantine/hooks';
import { PageHead } from '@dental/ui';
import { IconPlus, IconSearch } from '@tabler/icons-react';
import { useMemo, useState } from 'react';
import { useTranslation } from 'react-i18next';
import { useAuthStore } from '@/stores/auth-store.ts';
import { useDeleteService, useServices } from '../services.api';
import { filterServices, groupByCategory } from '../services.mappers';
import type { ServiceVM } from '../services.types';
import { ServiceFormModal } from './ServiceFormModal';
import { ServicesTable } from './ServicesTable';

export function ServicesScreen() {
  const { t } = useTranslation();
  const { hasPermission } = useAuthStore();
  const canManage = hasPermission('service.manage');

  const services = useServices();
  const deleteService = useDeleteService();

  const [search, setSearch] = useState('');
  const [category, setCategory] = useState<string | null>(null);
  const [editing, setEditing] = useState<Service | null>(null);
  const [pendingDelete, setPendingDelete] = useState<ServiceVM | null>(null);
  const [formOpened, { open: openForm, close: closeForm }] = useDisclosure(false);

  const list = useMemo(() => services.data ?? [], [services.data]);

  const categoryOptions = useMemo(() => {
    const set = new Set<string>();
    for (const s of list) if (s.category) set.add(s.category);
    return [...set].sort((a, b) => a.localeCompare(b)).map((c) => ({ value: c, label: c }));
  }, [list]);

  const groups = useMemo(
    () => groupByCategory(filterServices(list, search, category)),
    [list, search, category],
  );

  const openCreate = () => {
    setEditing(null);
    openForm();
  };
  const openEdit = (id: string) => {
    setEditing(list.find((s) => s.id === id) ?? null);
    openForm();
  };
  const handleFormClose = () => {
    setEditing(null);
    closeForm();
  };
  const confirmDelete = () => {
    if (pendingDelete) deleteService.mutate(pendingDelete.id);
    setPendingDelete(null);
  };

  return (
    <Container size="xl">
      <Stack gap="lg">
        <PageHead
          title={t('services.title')}
          actions={
            canManage && (
              <Button leftSection={<IconPlus size={18} />} onClick={openCreate}>
                {t('services.create')}
              </Button>
            )
          }
        />

        <Group>
          <TextInput
            leftSection={<IconSearch size={16} />}
            placeholder={t('services.search')}
            value={search}
            onChange={(e) => setSearch(e.currentTarget.value)}
            flex={1}
          />
          <Select
            placeholder={t('services.category')}
            data={categoryOptions}
            value={category}
            onChange={setCategory}
            clearable
            w={220}
          />
        </Group>

        {list.length === 0 && <Text c="dimmed">{t('services.empty')}</Text>}
        {list.length > 0 && (
          <ServicesTable
            groups={groups}
            canManage={canManage}
            onEdit={openEdit}
            onDelete={setPendingDelete}
          />
        )}
      </Stack>

      <ServiceFormModal opened={formOpened} onClose={handleFormClose} editing={editing} />

      <Modal
        opened={!!pendingDelete}
        onClose={() => setPendingDelete(null)}
        title={t('services.confirm_delete')}
        size="sm"
      >
        <Stack gap="md">
          <Text size="sm">
            {pendingDelete?.nameUz} ({pendingDelete?.code})
          </Text>
          <Group justify="flex-end">
            <Button variant="default" onClick={() => setPendingDelete(null)}>
              {t('common.cancel')}
            </Button>
            <Button color="red" loading={deleteService.isPending} onClick={confirmDelete}>
              {t('common.delete')}
            </Button>
          </Group>
        </Stack>
      </Modal>
    </Container>
  );
}
