`updateStatus` funksiyasini o'zgartiring. Mavjud `db.update(queueTickets)...returning()` dan **keyin**, audit'dan **oldin**, status sync qo'shing:

```typescript
  async function updateStatus(c: Context, id: string, input: UpdateQueueStatusInput) {
    const db = getDb(c);
    const [before] = await db.select().from(queueTickets).where(eq(queueTickets.id, id));
    if (!before) return null;

    const now = new Date();
    const patch: Record<string, unknown> = { status: input.status };
    if (input.doctorId) patch.doctorId = input.doctorId;
    if (input.status === 'in_progress') patch.calledAt = now;
    if (input.status === 'done' || input.status === 'missed') patch.finishedAt = now;

    const [updated] = await db
      .update(queueTickets)
      .set(patch)
      .where(eq(queueTickets.id, id))
      .returning();

    // Appointment status sync (faqat bog'langan ticketlarda)
    if (updated && before.appointmentId) {
      const apptStatus =
        input.status === 'in_progress'
          ? ('in_progress' as const)
          : input.status === 'done'
            ? ('completed' as const)
            : input.status === 'missed'
              ? ('no_show' as const)
              : null;

      if (apptStatus) {
        const [appt] = await db
          .select({ id: appointments.id, status: appointments.status })
          .from(appointments)
          .where(eq(appointments.id, before.appointmentId));

        // Terminal holatni qayta yozmaslik
        if (appt && appt.status !== 'cancelled' && appt.status !== 'completed') {
          await db
            .update(appointments)
            .set({ status: apptStatus, updatedAt: now })
            .where(eq(appointments.id, before.appointmentId));

          await deps.audit.record(c, {
            action: 'status_synced',
            resourceType: 'appointment',
            resourceId: before.appointmentId,
            before: { status: appt.status },
            after: { status: apptStatus },
          });
        }
      }
    }

    if (updated) {
      await deps.audit.record(c, {
        action: 'status_changed',
        resourceType: 'queue_ticket',
        resourceId: id,
        before: { status: before.status },
        after: { status: input.status },
      });
    }
    return updated;
  }
```

- [ ] **Step 4: `checkInCandidates` funksiyasi**

`updateStatus` dan keyin, `return {...}` dan oldin qo'shing:

```typescript
  /** Bugungi scheduled/confirmed appointment'lar, navbatga olinmaganlari. */
  async function checkInCandidates(c: Context, branchId: string) {
    const db = getDb(c);
    const todayStart = new Date();
    todayStart.setUTCHours(0, 0, 0, 0);
    const todayEnd = new Date(todayStart);
    todayEnd.setUTCDate(todayEnd.getUTCDate() + 1);

    return db
      .select({
        id: appointments.id,
        tenantId: appointments.tenantId,
        branchId: appointments.branchId,
        patientId: appointments.patientId,
        doctorId: appointments.doctorId,
        assistantId: appointments.assistantId,
        startTime: appointments.startTime,
        endTime: appointments.endTime,
        status: appointments.status,
        serviceIds: appointments.serviceIds,
        notes: appointments.notes,
        cancellationReason: appointments.cancellationReason,
        createdAt: appointments.createdAt,
        updatedAt: appointments.updatedAt,
      })
      .from(appointments)
      .leftJoin(queueTickets, eq(queueTickets.appointmentId, appointments.id))
      .where(
        and(
          eq(appointments.branchId, branchId),
          gte(appointments.startTime, todayStart),
          lt(appointments.startTime, todayEnd),
          isNull(queueTickets.id),
          sql`${appointments.status} in ('scheduled','confirmed')`,
        ),
      )
      .orderBy(asc(appointments.startTime));
  }
```

```typescript
import type { QueueTicket } from '@dental/api-client';
import type { QueueCardVM, QueueColumnKey, QueueColumnVM, QueueLookups } from './queue.types';

export function toQueueCardVM(ticket: QueueTicket, lookups: QueueLookups): QueueCardVM {
  return {
    id: ticket.id,
    queueNumber: ticket.queueNumber,
    status: ticket.status,
    patientName: ticket.patientId
      ? (lookups.patientNameById.get(ticket.patientId) ?? '—')
      : '—',
    serviceName: ticket.serviceId
      ? (lookups.serviceNameById.get(ticket.serviceId) ?? null)
      : null,
    createdAt: ticket.createdAt,
    calledAt: ticket.calledAt,
  };
}

const COLUMN_OF: Record<QueueTicket['status'], QueueColumnKey> = {
  waiting: 'waiting',
  in_progress: 'in_progress',
  done: 'completed',
  missed: 'completed',
};

const COLUMN_ORDER: QueueColumnKey[] = ['waiting', 'in_progress', 'completed'];

export function groupByStatus(tickets: QueueTicket[]): QueueColumnVM[] {
  return COLUMN_ORDER.map((key) => ({
    key,
    tickets: tickets.filter((t) => COLUMN_OF[t.status] === key),
  }));
}
```

```tsx
import type { QueueCardVM } from '../queue.types';

const statusColor: Record<QueueStatus, string> = {
  waiting: 'yellow',
  in_progress: 'blue',
  done: 'teal',
  missed: 'red',
};

function hhmm(iso: string) {
  return new Date(iso).toLocaleTimeString('uz-UZ', { hour: '2-digit', minute: '2-digit' });
}

export function QueueCard({
  vm,
  canManage,
  onStatus,
}: {
  vm: QueueCardVM;
  canManage: boolean;
  onStatus: (status: QueueStatus) => void;
}) {
  const { t } = useTranslation();
  return (
    <Card withBorder padding="sm" radius="sm">
      <Group justify="space-between" wrap="nowrap">
        <Stack gap={2}>
          <Group gap="xs">
            <Badge size="lg" color={statusColor[vm.status]}>
              #{vm.queueNumber}
            </Badge>
            <Text size="sm">{vm.patientName}</Text>
          </Group>
          {vm.serviceName && (
            <Text size="xs" c="dimmed">
              {vm.serviceName}
            </Text>
          )}
          <Text size="xs" c="dimmed">
            {vm.status === 'in_progress' && vm.calledAt
              ? `${t('queue.called_at')} ${hhmm(vm.calledAt)}`
              : hhmm(vm.createdAt)}
          </Text>
        </Stack>
        {canManage && vm.status === 'waiting' && (
          <ActionIcon color="blue" onClick={() => onStatus('in_progress')} aria-label={t('queue.call')}>
            <IconPlayerPlay size={16} />
          </ActionIcon>
        )}
        {canManage && vm.status === 'in_progress' && (
          <Group gap="xs" wrap="nowrap">
            <Button size="xs" color="teal" onClick={() => onStatus('done')}>
              {t('queue.done')}
            </Button>
            <ActionIcon color="red" variant="subtle" onClick={() => onStatus('missed')} aria-label={t('queue.missed')}>
              <IconX size={16} />
            </ActionIcon>
          </Group>
        )}
      </Group>
    </Card>
  );
}
```

```tsx
export function QueueBoard() {
  const { t } = useTranslation();
  const { hasPermission, activeBranchId } = useAuthStore();
  const [branchFilter, setBranchFilter] = useState<string | null>(activeBranchId);
  const [opened, { open, close }] = useDisclosure(false);

  const services = useServices();
  const queue = useQueueToday(branchFilter);
  const updateStatus = useUpdateQueueStatus();

  const lookups: QueueLookups = useMemo(
    () => ({
      patientNameById: new Map(
        (patients.data?.data ?? []).map((p) => [p.id, p.fullName] as const),
      ),
      serviceNameById: new Map((services.data ?? []).map((s) => [s.id, s.nameUz] as const)),
    }),
    [patients.data, services.data],
  );

  const columns = useMemo(() => groupByStatus(queue.data ?? []), [queue.data]);
  const canManage = hasPermission('queue.manage');

  const onStatus = (id: string, status: QueueStatus) => updateStatus.mutate({ id, status });

  return (
    <Container size="xl">
      <Stack gap="lg">
        <Group justify="space-between">
          <Title order={2}>{t('queue.title')}</Title>
          <Group>
            <Select
              data={branches.data?.map((b) => ({ value: b.id, label: b.name })) ?? []}
              value={branchFilter}
              onChange={setBranchFilter}
              placeholder={t('queue.select_branch')}
            />
            {hasPermission('queue.create') && (
              <Button leftSection={<IconPlus size={18} />} onClick={open} disabled={!branchFilter}>
                {t('queue.new_ticket')}
              </Button>
            )}
          </Group>
        </Group>

        {!branchFilter && <Text c="dimmed">{t('queue.select_branch_hint')}</Text>}

        {branchFilter && (
          <Group align="flex-start" grow>
            {columns.map((column) => (
              <QueueColumn
                key={column.key}
                column={column}
                title={t(titleKey[column.key])}
                lookups={lookups}
                canManage={canManage}
                onStatus={onStatus}
              />
            ))}
          </Group>
        )}
      </Stack>

      {branchFilter && (
        <CheckInDialog opened={opened} onClose={close} branchId={branchFilter} />
      )}
    </Container>
  );
}
```

```tsx
import { Button, Group, Modal, Select, Stack, Tabs, Text } from '@mantine/core';
import { useState } from 'react';
import { useTranslation } from 'react-i18next';
import { useCheckInCandidates, useCreateTicket, usePatientsAll, useServices } from '../queue.api';

export function CheckInDialog({
  opened,
  onClose,
  branchId,
}: {
  opened: boolean;
  onClose: () => void;
  branchId: string;
}) {
  const { t } = useTranslation();
  const candidates = useCheckInCandidates(branchId);
  const patients = usePatientsAll();
  const services = useServices();
  const createTicket = useCreateTicket();

  const [walkInPatient, setWalkInPatient] = useState<string | null>(null);
  const [walkInService, setWalkInService] = useState<string | null>(null);

  function hhmm(iso: string) {
    return new Date(iso).toLocaleTimeString('uz-UZ', { hour: '2-digit', minute: '2-digit' });
  }

  const handleClose = () => {
    setWalkInPatient(null);
    setWalkInService(null);
    onClose();
  };

  const checkIn = (appointmentId: string, patientId: string, serviceId?: string) => {
    createTicket.mutate(
      { branchId, appointmentId, patientId, serviceId },
      { onSuccess: handleClose },
    );
  };

  const walkIn = () => {
    createTicket.mutate(
      {
        branchId,
        patientId: walkInPatient ?? undefined,
        serviceId: walkInService ?? undefined,
      },
      { onSuccess: handleClose },
    );
  };

  const patientName = (id: string) =>
    patients.data?.data.find((p) => p.id === id)?.fullName ?? '—';

  return (
    <Modal opened={opened} onClose={handleClose} title={t('queue.new_ticket')}>
      <Tabs defaultValue="from_appointment">
        <Tabs.List mb="md">
          <Tabs.Tab value="from_appointment">{t('queue.from_appointment')}</Tabs.Tab>
          <Tabs.Tab value="walk_in">{t('queue.walk_in')}</Tabs.Tab>
        </Tabs.List>

        <Tabs.Panel value="from_appointment">
          <Stack gap="sm">
            {(candidates.data ?? []).length === 0 && (
              <Text c="dimmed" size="sm">
                {t('queue.no_candidates')}
              </Text>
            )}
            {(candidates.data ?? []).map((a) => (
              <Group key={a.id} justify="space-between">
                <Stack gap={0}>
                  <Text size="sm">{patientName(a.patientId)}</Text>
                  <Text size="xs" c="dimmed">
                    {hhmm(a.startTime)}
                  </Text>
                </Stack>
                <Button
                  size="xs"
                  onClick={() => checkIn(a.id, a.patientId, a.serviceIds?.[0])}
                  loading={createTicket.isPending}
                >
                  {t('queue.check_in')}
                </Button>
              </Group>
            ))}
          </Stack>
        </Tabs.Panel>

        <Tabs.Panel value="walk_in">
          <Stack gap="md">
            <Select
              label={t('queue.patient')}
              data={patients.data?.data.map((p) => ({ value: p.id, label: p.fullName })) ?? []}
              searchable
              clearable
              value={walkInPatient}
              onChange={setWalkInPatient}
            />
            <Select
              label={t('queue.service')}
              data={
                services.data
                  ?.filter((s) => s.isActive)
                  .map((s) => ({ value: s.id, label: s.nameUz })) ?? []
              }
              searchable
              clearable
              value={walkInService}
              onChange={setWalkInService}
            />
            <Group justify="flex-end">
              <Button variant="default" onClick={handleClose}>
                {t('common.cancel')}
              </Button>
              <Button onClick={walkIn} loading={createTicket.isPending}>
                {t('queue.add')}
              </Button>
            </Group>
          </Stack>
        </Tabs.Panel>
      </Tabs>
    </Modal>
  );
}
```
