function isVisible(item: NavItem, has: (p: string) => boolean): boolean {
  if (item.disabled) return true;
  if (!item.permission) return true;
  const perms = Array.isArray(item.permission) ? item.permission : [item.permission];
  return perms.some((p) => has(p));

  return NAV_GROUPS

    .filter((g) => g.items.length > 0);
}

- [ ] **Step 4: `Sidebar.tsx`**

```tsx
import { Link, useLocation } from 'react-router-dom';
import { useTranslation } from 'react-i18next';
import { useAuthStore } from '@/stores/auth-store.ts';
import { visibleGroups, type NavItem } from './nav-config';
import classes from './Sidebar.module.css';

function isActive(pathname: string, path: string): boolean {
  if (path === '/') return pathname === '/';
  return pathname === path || pathname.startsWith(`${path}/`);
}

export function Sidebar() {
  const { t } = useTranslation();
  const { hasPermission } = useAuthStore();
  const { pathname } = useLocation();
  const groups = visibleGroups(hasPermission);

  const renderItem = (item: NavItem) => {
    const Icon = item.icon;
    const label = t(item.labelKey);
    if (item.disabled) {
      return (
        <span key={item.key} className={classes.item} aria-disabled="true" title={t('shell.coming_soon')}>
          <Icon />
          {label}
          <span className={classes.soon}>{t('shell.coming_soon')}</span>
        </span>
      );
    }
    return (
      <Link key={item.key} to={item.path} className={classes.item} data-active={isActive(pathname, item.path)}>
        <Icon />
        {label}
      </Link>
    );
  };

  return (
    <>
      <div className={classes.top}>
        <span className={classes.topLabel}>{t('app.name')}</span>
        <span className={classes.pin} />
      </div>
      <nav className={classes.scroll}>
        {groups.map((g) => (
          <div key={g.labelKey} className={classes.group}>
            <div className={classes.groupLabel}>{t(g.labelKey)}</div>
            {g.items.map(renderItem)}
          </div>
        ))}
      </nav>
    </>
  );
}
```
> `app.name` mavjud i18n kalit (clinik filial nomi o'rniga — `sb-top` aniq filial nomi manbai plan ochiq nuqtasi; hozir tenant/app nomi). Active `data-active` boolean → CSS `[data-active="true"]`.

```tsx
import { Burger, Menu, useMantineColorScheme } from '@mantine/core';
import { Search, Bell, Sun, Moon, LogOut, User, Repeat } from 'lucide-react';
import { useTranslation } from 'react-i18next';
import { useNavigate } from 'react-router-dom';
import { IconButton, Avatar } from '@dental/ui';
import { api } from '@/lib/api.ts';
import { useAuthStore } from '@/stores/auth-store.ts';
import classes from './Topbar.module.css';

export function Topbar({ onBurger, burgerOpened }: { onBurger: () => void; burgerOpened: boolean }) {
  const { t, i18n } = useTranslation();
  const navigate = useNavigate();
  const { colorScheme, toggleColorScheme } = useMantineColorScheme();
  const { user, refreshToken, clear } = useAuthStore();

  const logout = async () => {
    if (refreshToken) {
      try { await api.auth.logout(refreshToken); } catch { /* silent */ }
    }
    clear();
    navigate('/login');
  };

  return (
    <div className={classes.bar}>
      <Burger opened={burgerOpened} onClick={onBurger} hiddenFrom="sm" size="sm" />

      <button type="button" className={classes.ws} onClick={() => navigate('/select-workspace')}>
        <span className={classes.wsLogo}>{t('app.name').charAt(0)}</span>
        <span className={classes.wsName}>
          <b>{t('app.name')}</b>
          <span>{user?.phone ?? ''}</span>
        </span>
      </button>

      <div className={classes.search} role="search" aria-label={t('shell.search_placeholder')}>
        <Search />
        <span style={{ flex: 1 }}>{t('shell.search_placeholder')}</span>
        <span className={classes.kbd}>⌘K</span>
      </div>

      <span className={classes.spacer} />

      <IconButton label={t(colorScheme === 'dark' ? 'shell.theme_light' : 'shell.theme_dark')} onClick={toggleColorScheme}>
        {colorScheme === 'dark' ? <Sun /> : <Moon />}
      </IconButton>

      <IconButton label={t('shell.notifications')} dot>
        <Bell />
      </IconButton>

      <Menu shadow="md" width={140}>
        <Menu.Target>
          <button type="button" className={classes.lang} aria-label="Language">
            <b>{i18n.language === 'ru' ? 'RU' : 'UZ'}</b>
            <span>{i18n.language === 'ru' ? 'UZ' : 'RU'}</span>
          </button>
        </Menu.Target>
        <Menu.Dropdown>
          <Menu.Item onClick={() => i18n.changeLanguage('uz')}>O'zbek</Menu.Item>
          <Menu.Item onClick={() => i18n.changeLanguage('ru')}>Русский</Menu.Item>
        </Menu.Dropdown>
      </Menu>

      <Menu shadow="md" width={220}>
        <Menu.Target>
          <button type="button" className={classes.prof}>
            <Avatar name={user?.fullName ?? '?'} size="md" />
            <span className={classes.profName}>
              <b>{user?.fullName ?? user?.phone}</b>
              <span>{user?.phone}</span>
            </span>
          </button>
        </Menu.Target>
        <Menu.Dropdown>
          <Menu.Item leftSection={<User size={16} />} onClick={() => navigate('/profile')}>
            {t('profile.title')}
          </Menu.Item>
          <Menu.Item leftSection={<Repeat size={16} />} onClick={() => navigate('/select-workspace')}>
            {t('layout.switch_workspace')}
          </Menu.Item>
          <Menu.Divider />
          <Menu.Item color="red" leftSection={<LogOut size={16} />} onClick={logout}>
            {t('layout.logout')}
          </Menu.Item>
        </Menu.Dropdown>
      </Menu>
    </div>
  );
}
```

```tsx
import type { ReactNode } from 'react';
import { Navigate } from 'react-router-dom';
import { useAuthStore } from '@/stores/auth-store.ts';

export function RequirePermission({ perm, children }: { perm: string | string[]; children: ReactNode }) {
  const { hasPermission } = useAuthStore();
  const perms = Array.isArray(perm) ? perm : [perm];
  const allowed = perms.some((p) => hasPermission(p));
  if (!allowed) return <Navigate to="/403" replace />;
  return <>{children}</>;
}
```
