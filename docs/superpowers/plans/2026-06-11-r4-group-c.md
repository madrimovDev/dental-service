```ts
import type { NotificationChannel } from './types.ts';
const BASE = 'https://notify.eskiz.uz/api';
export const createEskizChannel = (config: { email: string; password: string }): NotificationChannel => {
  let token: string | null = null;
  async function auth(): Promise<string> {
    if (token) return token;
    const res = await fetch(`${BASE}/auth/login`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email: config.email, password: config.password }),
    });
    const json = (await res.json()) as { data?: { token?: string } };
    token = json.data?.token ?? null;
    if (!token) throw new Error('Eskiz auth failed');
    return token;
  }
  return {
    async send(to, message) {
      try {
        const t = await auth();
        const res = await fetch(`${BASE}/message/sms/send`, {
          method: 'POST',
          headers: { Authorization: `Bearer ${t}`, 'Content-Type': 'application/json' },
          body: JSON.stringify({ mobile_phone: to.replace(/\D/g, ''), message: message.text, from: '4546' }),
        });
        if (!res.ok) return { ok: false, error: `Eskiz ${res.status}` };
        const json = (await res.json()) as { id?: string };
        return { ok: true, providerId: json.id };
      } catch (e) {
        token = null;
        return { ok: false, error: e instanceof Error ? e.message : 'eskiz_error' };
      }
    },
  };
};
```

/** Eslatma kechikishi (ms). start − leadHours dan now gacha. O'tgan/juda yaqin → null (enqueue qilinmaydi). */
export function reminderDelayMs(startTime: Date, now: Date, leadHours = 24): number | null {
  const fireAt = startTime.getTime() - leadHours * 3600_000;
  const delay = fireAt - now.getTime();
  return delay > 0 ? delay : null;
}

- [ ] **Step 4: reminder-worker.ts**
```ts
import { Worker } from 'bullmq';
import { appointments, patients } from '@dental/db-schema';
import { eq } from 'drizzle-orm';
import { env } from './env.ts';
import { withTenant } from '../db/scoped.ts';
import { createSmsChannel } from '@dental/notifications';
import { createNotificationsService } from '../modules/notifications/index.ts';
import type { ReminderJob } from './queue.ts';

export function startReminderWorker() {
  const sms = createSmsChannel({ eskizEmail: env.ESKIZ_EMAIL, eskizPassword: env.ESKIZ_PASSWORD });
  const svc = createNotificationsService({ sms });
  return new Worker<ReminderJob>('appointment-reminder', async (job) => {
    const { tenantId, appointmentId } = job.data;
    await withTenant(tenantId, async (tx) => {
      const [appt] = await tx.select().from(appointments).where(eq(appointments.id, appointmentId));
      if (!appt) return;
      const [patient] = await tx.select().from(patients).where(eq(patients.id, appt.patientId));
      await svc.notifyInTx(tx, tenantId, {
        userId: appt.doctorId,
        type: 'appointment_reminder',
        title: 'Yozilish eslatmasi',
        body: `Bemor: ${patient?.fullName ?? ''}, ${appt.startTime.toISOString()}`,
        smsTo: patient?.phone ?? undefined,
      });
    });
  }, { connection: { url: env.REDIS_URL } });
}
```
